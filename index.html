// ✅ ADD THIS TO YOUR WEBSITE — replaces the checkForOrderSuccess() function at the bottom of index.html // Show a clear thank-you confirmation after a successful Stripe payment, // since customers were previously landing back on a normal-looking // homepage with no acknowledgment their order went through at all. (function checkForOrderSuccess() { const params = new URLSearchParams(window.location.search); if (params.get("order") !== "success") return; // ✅ FIRE GA4 PURCHASE EVENT IMMEDIATELY // Parse the session_id from Stripe's redirect (your API server must add this) const sessionId = params.get("session_id"); if (sessionId && window.gtag) { // Fetch Stripe session details to get order total & items fetch(`${API_BASE}/api/stripe/session/${sessionId}`) .then(res => res.json()) .then(session => { if (session && session.amount_total) { // Build items array from line items const items = (session.line_items?.data || []).map(line => ({ item_id: line.price?.product?.id || "unknown", item_name: line.price?.product?.name || "Photo Gift", item_category: "photo_gift", price: (line.price?.unit_amount / 100).toFixed(2), quantity: line.quantity || 1, })); window.gtag("event", "purchase", { transaction_id: sessionId, affiliation: "onjjem.com", value: (session.amount_total / 100).toFixed(2), currency: session.currency?.toUpperCase() || "GBP", tax: 0, shipping: 0, items: items.length > 0 ? items : [ { item_id: "photo-gift", item_name: "Personalised Photo Gift", price: (session.amount_total / 100).toFixed(2), quantity: 1, }, ], }); } }) .catch(err => console.warn("Could not fetch session details for GA4:", err)); } const overlay = document.createElement("div"); overlay.className = "thankyou-overlay"; overlay.innerHTML = `
✓
Thank you for your order!
Your payment has gone through successfully. We're getting your order ready to print now, and it'll be on its way to you soon. If you have any questions in the meantime, feel free to get in touch.
`; document.body.appendChild(overlay); overlay.querySelector(".thankyou-close").addEventListener("click", () => { overlay.remove(); // clean the ?order=success param out of the URL so a refresh doesn't re-show this const cleanUrl = window.location.origin + window.location.pathname; window.history.replaceState({}, "", cleanUrl); }); })();
